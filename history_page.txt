+function defaultStartIso(): string {
+  return formatISO(subMonths(startOfMonth(new Date()), 12), { representation: 'date' })
+}
+function defaultEndIso(): string {
+  return formatISO(endOfMonth(new Date()), { representation: 'date' })

+// ============================================================================
+// HELPER 1 · resolveFilters (Paquete 7.1 · extraído de cuerpo)
+// ============================================================================
+
+function resolveFilters(initial: RawParsedDireccionFilters): ResolvedDireccionFilters {
+  const startDateRaw = initial.startDate
+  const endDateRaw = initial.endDate
+  const startDate =
+    typeof startDateRaw === 'string' && startDateRaw.length === 10 ? startDateRaw : defaultStartIso()
+  const endDate =
+    typeof endDateRaw === 'string' && endDateRaw.length === 10 ? endDateRaw : defaultEndIso()
+  const defaultSections = DIRECCION_SECTIONS.map(s => s.id) as DireccionSectionId[]
+  return {
+    ...initial,
+    startDate,
+    endDate,
+    companyId: initial.companyId,
+    sections: initial.sections?.length ? (initial.sections as DireccionSectionId[]) : defaultSections,
+  }
+}

+// ============================================================================
+// HELPER 4 · lookupCompanyMetadata (Paquete 7.1 · Extract prisma try/catch L148-161)
+// ============================================================================
+
+type CompanyMeta = Readonly<{ rfc: string | null; displayName: string | null }>
+
+async function lookupCompanyMetadata(companyId: string | undefined): Promise<CompanyMeta> {
+  if (!companyId) return { rfc: null, displayName: null }
+  try {
+    const company = await prisma.company.findUnique({
+      where: { id: companyId },
+      select: { rfc: true, businessName: true, name: true },
+    })
+    if (!company) return { rfc: null, displayName: null }
+    return {
+      rfc: company.rfc,
+      displayName: company.businessName || company.name || null,
+    }
+  } catch {
+    return { rfc: null, displayName: null }
+  }
+}

+    fiscalEntityId: filters.fiscalEntityId ?? filters.companyId,
+    startDate: filters.startDate,
+    endDate: filters.endDate,
+    departamento: filters.departamento ?? null,
+  }

+function renderSectionsOrEmpty(
+  filters: ResolvedDireccionFilters,
+  scope: DireccionFetchScope,
+  view: DireccionViewMode,
+  anioComparativo: number | null,
+): ReactNode {

+  return (
+    <>
+      {secs.includes('kpis') && (
+        <Suspense key="sec-kpis" fallback={<KpiGridSkeleton />}>
+          <DireccionKpisSection scope={scope} />
+        </Suspense>
+      )}
+      {secs.includes('tendencia_historica') && (
+        <Suspense key="sec-tendencia" fallback={<ChartSkeleton label="Tendencia Histórica y YoY" />}>
+          <DireccionTendenciaChart scope={scope} view={view} anioComparativo={anioComparativo} />

+// ============================================================================
+
+function renderBody(
+  filters: ResolvedDireccionFilters,
+  scope: DireccionFetchScope | null,
+): ReactNode {
+  if (!filters.companyId || !scope) return <CardSeleccionaEmpresa />

+// ============================================================================
+// COMPONENTE PRINCIPAL SERVER PAGE (Paquete 7.1 · ORQUESTADOR <50 líneas)
+// ============================================================================
+
+export async function DireccionDashboardServerPage({
+  initialFilters,
+}: Readonly<{ initialFilters: ReturnType<typeof parseDireccionSearchParams> }>) {
+  const filters = resolveFilters(initialFilters)
+  const hasCompany = !!filters.companyId && filters.companyId.length > 0
+  const filtersForFilterBar = buildFiltersForFilterBar(filters)
+
+  const { rfc: companyRfc, displayName: companyDisplayName } = hasCompany
+    ? await lookupCompanyMetadata(filters.companyId)
+    : { rfc: null as string | null, displayName: null as string | null }
+
+  const scope = resolveFetchScope(filters)
+
+  return (
+    <ProtectedRoute>
+      <div className="flex-1 space-y-4 p-4 md:p-6 pt-6">
+        <HeaderDireccion
+          filters={filters}
+          companyRfc={companyRfc}
+          companyDisplayName={companyDisplayName}
+        />
+        <div className="flex flex-col sm:flex-row gap-4 py-4 items-end">
+          <DireccionFilterBar filters={filtersForFilterBar} />
+        </div>
+        {renderBody(filters, scope)}
+      </div>
+    </ProtectedRoute>
+  )
+}
